cmake_minimum_required(VERSION 3.22)
project(MothDSPLab VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
include(FetchContent)
set(JUCE_SOURCE_DIR "" CACHE PATH "Existing JUCE checkout (optional)")
if(JUCE_SOURCE_DIR)
  add_subdirectory("${JUCE_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/juce")
else()
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG f72bad64d29715216226685810c5196bd0d79d77 GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(JUCE)
endif()
add_library(lab_core STATIC
  src/measurement/MeasurementReport.cpp src/ai-bridge/DeviceProfile.cpp src/ai-bridge/Scenario.cpp src/core/SessionStore.cpp src/analysis/AnalysisEngine.cpp
  src/ui/Plot.cpp src/measurement/GuidedWorkflow.cpp src/measurement/MeasurementEngine.cpp src/audio-host/PluginHost.cpp src/hardware/SetupTest.cpp src/measurement/HardwareComparison.cpp)
target_include_directories(lab_core PUBLIC src)
target_compile_definitions(lab_core PRIVATE LAB_DEVELOPMENT_BUILD_DIR="${CMAKE_BINARY_DIR}" LAB_DEVELOPMENT_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
target_compile_definitions(lab_core PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_PLUGINHOST_VST3=1 JUCE_PLUGINHOST_AU=$<BOOL:${APPLE}> JUCE_MODAL_LOOPS_PERMITTED=1)
target_link_libraries(lab_core PUBLIC juce::juce_audio_utils juce::juce_dsp juce::juce_cryptography)
juce_add_gui_app(MothDSPLab PRODUCT_NAME "Moth DSP Lab" COMPANY_NAME "Moth Production" BUNDLE_ID "hu.mothproduction.dsplab"
  MICROPHONE_PERMISSION_ENABLED TRUE MICROPHONE_PERMISSION_TEXT "Record the selected audio-interface input to check your pedal signal path.")
target_sources(MothDSPLab PRIVATE src/ui/Main.cpp src/ui/MainComponent.cpp src/ui/HardwareSetupComponent.cpp)
target_link_libraries(MothDSPLab PRIVATE lab_core)
if(APPLE)
  target_sources(MothDSPLab PRIVATE src/ui/AudioPermission_mac.mm)
  target_link_libraries(MothDSPLab PRIVATE "-framework AVFoundation")
else()
  target_sources(MothDSPLab PRIVATE src/ui/AudioPermission.cpp)
endif()
if(BUILD_TESTING)
  juce_add_console_app(lab_tests PRODUCT_NAME "Lab Tests")
  target_sources(lab_tests PRIVATE tests/CoreTests.cpp tests/DeviceProfileTests.cpp tests/HardwareTests.cpp tests/WorkflowTests.cpp tests/ScenarioTests.cpp)
  target_link_libraries(lab_tests PRIVATE lab_core)
  add_test(NAME lab_core_tests COMMAND lab_tests)
endif()
if(BUILD_TESTING)
  juce_add_plugin(LabFixture COMPANY_NAME "Moth Production" PLUGIN_MANUFACTURER_CODE Moth PLUGIN_CODE Mlft BUNDLE_ID "hu.mothproduction.dsplab.fixture"
    FORMATS VST3 PRODUCT_NAME "Moth Lab Fixture" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
    COPY_PLUGIN_AFTER_BUILD FALSE VST3_CAN_REPLACE_VST2 FALSE)
  target_sources(LabFixture PRIVATE tests/FixturePlugin.cpp)
  target_compile_definitions(LabFixture PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  target_link_libraries(LabFixture PRIVATE juce::juce_audio_processors)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME lab_vst3_integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py"
      "$<TARGET_FILE:MothDSPLab>" "$<TARGET_FILE:LabFixture_VST3>")
    set_tests_properties(lab_vst3_integration PROPERTIES TIMEOUT 120)
  endif()
endif()

if(BUILD_TESTING AND APPLE AND Python3_FOUND)
  add_test(NAME lab_au_integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/au_integration.py"
    "$<TARGET_FILE:MothDSPLab>" "$<TARGET_FILE:LabFixture_VST3>")
  set_tests_properties(lab_au_integration PROPERTIES TIMEOUT 120)
endif()
if(BUILD_TESTING AND Python3_FOUND)
  add_test(NAME lab_hardware_comparison COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/hardware_comparison.py"
    "$<TARGET_FILE:MothDSPLab>" "$<TARGET_FILE:LabFixture_VST3>")
  set_tests_properties(lab_hardware_comparison PROPERTIES TIMEOUT 120)
endif()

if(BUILD_TESTING)
  juce_add_console_app(lab_workflow_ui_tests PRODUCT_NAME "Lab Workflow UI Tests")
  target_sources(lab_workflow_ui_tests PRIVATE tests/WorkflowUiTests.cpp src/ui/HardwareSetupComponent.cpp)
  target_link_libraries(lab_workflow_ui_tests PRIVATE lab_core)
  if(APPLE)
    target_sources(lab_workflow_ui_tests PRIVATE src/ui/AudioPermission_mac.mm)
    target_link_libraries(lab_workflow_ui_tests PRIVATE "-framework AVFoundation")
  else()
    target_sources(lab_workflow_ui_tests PRIVATE src/ui/AudioPermission.cpp)
  endif()
  add_test(NAME lab_workflow_ui COMMAND lab_workflow_ui_tests)
  set_tests_properties(lab_workflow_ui PROPERTIES TIMEOUT 60)
endif()
